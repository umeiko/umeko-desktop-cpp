cmake_minimum_required(VERSION 3.24)
project(UmekoCpp VERSION 0.4.0 LANGUAGES C CXX)
if(APPLE)
  # The notification library also links into targets declared in this scope.
  enable_language(OBJCXX)
endif()
set(UMEKO_COMPILER_NAME "${CMAKE_CXX_COMPILER_ID}")
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  set(UMEKO_COMPILER_NAME "GCC")
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
  set(UMEKO_COMPILER_NAME "Apple Clang")
endif()
if(MINGW)
  set(UMEKO_COMPILER_NAME "MinGW ${UMEKO_COMPILER_NAME}")
endif()
set(UMEKO_COMPILER "${UMEKO_COMPILER_NAME} ${CMAKE_CXX_COMPILER_VERSION}")
option(UMEKO_BUNDLE_PYTHON "Include a standalone Python interpreter in release packages" ON)
if(UMEKO_BUNDLE_PYTHON)
  set(UMEKO_BUNDLE_PYTHON_JSON true)
else()
  set(UMEKO_BUNDLE_PYTHON_JSON false)
endif()
configure_file(app/build-config.json.in build-config.json @ONLY)
set(CMAKE_INSTALL_BINDIR "." CACHE PATH "Portable executable directory" FORCE)
set(QT_DEPLOY_BIN_DIR "." CACHE PATH "Portable executable directory" FORCE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Svg Quick QuickControls2 QuickDialogs2 Network Concurrent Test)
qt_standard_project_setup(REQUIRES 6.8)
include(FetchContent)
set(BUILD_WITH_QT6 ON CACHE BOOL "" FORCE)
set(BUILD_TEST_APPLICATION OFF CACHE BOOL "" FORCE)
set(BUILD_TRANSLATIONS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(WIN32)
  set(USE_CREDENTIAL_STORE OFF CACHE BOOL "Use DPAPI without the Credential Manager payload limit" FORCE)
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(qtkeychain GIT_REPOSITORY https://github.com/frankosterfeld/qtkeychain.git GIT_TAG ad7344c45a86a4f66cbafc4b081b5f7b876cb0b7)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(miniz GIT_REPOSITORY https://github.com/richgel999/miniz.git GIT_TAG 293d4db1b7d0ffee9756d035b9ac6f7431ef8492)
FetchContent_MakeAvailable(qtkeychain miniz)
set(APP_SOURCES src/common.cpp src/store.cpp src/chatmodel.cpp src/syntaxhighlighter.cpp src/backend.cpp src/sessions.cpp src/session_preview.cpp src/providers.cpp src/management.cpp src/about.cpp src/themes.cpp src/preferences.cpp src/common_providers.cpp src/update.cpp src/about_install.cpp)
set(UMEKO_NOTIFY_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory(native/notifications)
qt_add_executable(UmekoCpp src/main.cpp src/tray.cpp src/notifications.cpp ${APP_SOURCES})
target_link_libraries(UmekoCpp PRIVATE umeko_notifications)
if(UNIX AND NOT APPLE)
  find_package(Qt6 REQUIRED COMPONENTS DBus)
  target_link_libraries(UmekoCpp PRIVATE Qt6::DBus)
  add_subdirectory(native/input-method)
  add_dependencies(UmekoCpp fcitx5platforminputcontextplugin-qt6)
endif()
if(WIN32)
  target_sources(UmekoCpp PRIVATE app/windows.rc)
  set_target_properties(UmekoCpp PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
if(APPLE)
  set_target_properties(UmekoCpp PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_BUNDLE_NAME "Umeko-Desktop" MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.umeiko.umeko-desktop-cpp" MACOSX_BUNDLE_ICON_FILE app.icns)
  set_source_files_properties(app/icons/app.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_sources(UmekoCpp PRIVATE app/icons/app.icns)
endif()
if(UNIX AND NOT APPLE)
  set_target_properties(UmekoCpp PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
endif()
target_include_directories(UmekoCpp PRIVATE src ${qtkeychain_SOURCE_DIR} ${qtkeychain_BINARY_DIR})
target_link_libraries(UmekoCpp PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Svg Qt6::Quick Qt6::QuickControls2 Qt6::QuickDialogs2 Qt6::Network Qt6::Concurrent qt6keychain miniz)
target_compile_definitions(UmekoCpp PRIVATE UMEKO_VERSION="${PROJECT_VERSION}" UMEKO_VERSION_RC=${PROJECT_VERSION_MAJOR},${PROJECT_VERSION_MINOR},${PROJECT_VERSION_PATCH},0 UMEKO_COMPILER="${UMEKO_COMPILER}" UMEKO_BUILD_SYSTEM="${CMAKE_SYSTEM_NAME}" UMEKO_BUILD_ARCH="${CMAKE_SYSTEM_PROCESSOR}")
if(MSVC)
  target_compile_options(UmekoCpp PRIVATE /utf-8 /W4)
else()
  target_compile_options(UmekoCpp PRIVATE -Wall -Wextra -Wpedantic)
endif()
file(GLOB UI_QML CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} app/qml/*.qml app/qml/*.js)
file(GLOB_RECURSE UI_IMAGES CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} app/icons/*.webp app/icons/*.svg)
set_source_files_properties(app/qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
set_source_files_properties(app/qml/I18n.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(UmekoCpp URI Umeko.App VERSION 1.0 QML_FILES ${UI_QML} RESOURCES ${UI_IMAGES} app/icons/yanami-loading.png app/icons/yanami-catch.png app/icons/yanami-push.png app/icons/yanami-pull.png app/icons/yanami-installer.png app/icons/yanami-installing.png app/qml/qmldir app/theme-prompt.txt app/provider-presets.json app/icons/providers/codex.png app/icons/providers/kimi.png app/icons/providers/glm.png NO_CACHEGEN)
include(GNUInstallDirs)
set(CMAKE_INSTALL_BINDIR ".")
install(TARGETS UmekoCpp BUNDLE DESTINATION . COMPONENT Runtime RUNTIME DESTINATION . COMPONENT Runtime)
qt_generate_deploy_qml_app_script(TARGET UmekoCpp OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script} COMPONENT Runtime)
enable_testing()
qt_add_executable(UmekoCoreTests tests/core_test.cpp src/common.cpp src/store.cpp src/chatmodel.cpp src/syntaxhighlighter.cpp)
target_include_directories(UmekoCoreTests PRIVATE src ${qtkeychain_SOURCE_DIR} ${qtkeychain_BINARY_DIR})
target_link_libraries(UmekoCoreTests PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Test qt6keychain)
if(MSVC)
  target_compile_options(UmekoCoreTests PRIVATE /utf-8)
endif()
add_test(NAME native_core COMMAND UmekoCoreTests -o -,txt -o "${CMAKE_BINARY_DIR}/native-core.txt,txt")
qt_add_executable(UmekoUiTests tests/ui_test.cpp ${APP_SOURCES})
target_include_directories(UmekoUiTests PRIVATE src ${qtkeychain_SOURCE_DIR} ${qtkeychain_BINARY_DIR})
target_compile_definitions(UmekoUiTests PRIVATE UMEKO_VERSION="${PROJECT_VERSION}" UMEKO_COMPILER="${UMEKO_COMPILER}" UMEKO_BUILD_SYSTEM="${CMAKE_SYSTEM_NAME}" UMEKO_BUILD_ARCH="${CMAKE_SYSTEM_PROCESSOR}")
target_link_libraries(UmekoUiTests PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::QuickDialogs2 Qt6::Network Qt6::Concurrent Qt6::Test qt6keychain miniz)
if(MSVC)
  target_compile_options(UmekoUiTests PRIVATE /utf-8)
endif()
add_test(NAME native_ui COMMAND UmekoUiTests -o -,txt -o "${CMAKE_BINARY_DIR}/native-ui.txt,txt")
set(APP_TEST_PLATFORM offscreen)
if(APPLE)
  # Apple Keychain completion runs on the Cocoa main dispatch queue.
  set(APP_TEST_PLATFORM cocoa)
endif()
qt_add_executable(UmekoUpdateTests tests/update_test.cpp src/common.cpp src/update.cpp)
target_include_directories(UmekoUpdateTests PRIVATE src)
target_link_libraries(UmekoUpdateTests PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Test miniz)
target_compile_definitions(UmekoUpdateTests PRIVATE UMEKO_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
if(MSVC)
  target_compile_options(UmekoUpdateTests PRIVATE /utf-8)
endif()
add_test(NAME native_update COMMAND UmekoUpdateTests)
qt_add_executable(UmekoTrayTests tests/tray_test.cpp src/tray.cpp src/notifications.cpp)
target_include_directories(UmekoTrayTests PRIVATE src)
target_link_libraries(UmekoTrayTests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::Test umeko_notifications)
if(UNIX AND NOT APPLE)
  target_link_libraries(UmekoTrayTests PRIVATE Qt6::DBus)
endif()
if(MSVC)
  target_compile_options(UmekoTrayTests PRIVATE /utf-8)
endif()
add_test(NAME native_tray COMMAND UmekoTrayTests)
set_tests_properties(native_core native_ui native_update native_tray PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} ENVIRONMENT "UMEKO_LANGUAGE=zh;QT_QUICK_BACKEND=software;QT_QUICK_CONTROLS_STYLE=Basic;QT_QPA_PLATFORM=${APP_TEST_PLATFORM}")
